qiime cutadapt trim-paired --i-demultiplexed-sequences 1.1.reads.qza --p-front-f ACGTGCCAGCMGCCGCGGTAA --p-front-r ACGGACTACHVGGGTWTCTAAT --o-trimmed-sequences 1.2.reads.qza --verbose

#Using DADA2 to analyze quality scores of 10 random samples
qiime demux summarize --i-data 1.4.reads.qza --o-visualization 1.4.qual_viz

================================================================================================================
#2. Denoising with DADA2. Using quality score visualizations, you can choose trunc-len-f and trunc-len-r (note: sequences < trunc-len in length are discarded!)
# The drop-off for the forward reads was not so bad, but there is a significant drop-off in quality for the reverse reads, so let's trim 10bp
================================================================================================================

qiime dada2 denoise-paired --i-demultiplexed-seqs 1.4.reads.qza --o-table 2.table --o-representative-sequences 2.rep_seqs.qza --o-denoising-stats 2.denoise_stats --p-trunc-len-f 180 --p-trunc-len-r 200 --p-trim-left-f 0 --p-trim-left-r 0 --p-n-threads 5 --p-min-fold-parent-over-abundance 6 --p-min-overlap 12



qiime feature-table tabulate-seqs --i-data 2.rep_seqs.qza --o-visualization 2.rep_seqs.qzv 
qiime feature-table summarize --i-table 2.table.qza --o-visualization 2.table_summary.qzv
